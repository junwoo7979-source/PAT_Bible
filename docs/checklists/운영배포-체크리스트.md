### 마이그레이션 스크립트 작성 계획

#### Step 1: 평문 비밀번호 감지
```javascript
// functions/migrate-passwords.js (생성 예정)
const admin = require('firebase-admin');

async function detectPlaintextPasswords(churchCode) {
  const db = admin.firestore();
  const familiesRef = db.collection('churches').doc(churchCode).collection('families');
  const snapshot = await familiesRef.get();
  
  const plaintext = [];
  snapshot.forEach(doc => {
    const data = doc.data();
    // v1$ prefix 없으면 평문
    if (data.familyPassword && !data.familyPassword.startsWith('v1$')) {
      plaintext.push({ id: doc.id, data });
    }
  });
  return plaintext;
}
```

#### Step 2: 해시로 변환 및 저장
```javascript
// 감지된 각 가족에 대해:
const { hashFamilyPassword } = require('./password');

async function migratePasswordsForChurch(churchCode, pepper) {
  const plaintext = await detectPlaintextPasswords(churchCode);
  
  for (const { id, data } of plaintext) {
    const hashed = hashFamilyPassword(churchCode, data.familyPassword, pepper);
    await db.collection('churches').doc(churchCode)
      .collection('families').doc(id)
      .update({ 
        familyPassword: hashed,
        migratedAt: admin.firestore.FieldValue.serverTimestamp()
      });
  }
  
  console.log(`✓ Migrated ${plaintext.length} families`);
}
```

#### Step 3: 운영 중 실행 명령

```bash
# 이후 관리자 대시보드에서 수동 실행
# (또는 Cloud Scheduler로 자동화)
```
